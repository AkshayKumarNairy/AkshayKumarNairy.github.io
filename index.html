import { test, expect } from '@playwright/test';
import { LoginPage } from './pages/LoginPage';
import { AdDashboard } from './pages/AdDashboard';

test.describe('Merchant Ad Workflow', () => {

  test.beforeEach(async ({ page }) => {
    const login = new LoginPage(page);
    await login.goto();
    await login.signIn(process.env.TEST_USER!, process.env.TEST_PASS!);
  });

  test('merchant can create and publish a listing', async ({ page }) => {
    const dashboard = new AdDashboard(page);

    await dashboard.goto();
    await dashboard.clickCreateAd();

    await page.fill('[data-testid="ad-title"]', 'Sample Listing Title');
    await page.fill('[data-testid="monthly-rate"]', '299');
    await page.selectOption('[data-testid="category"]', 'standard');

    await page.click('[data-testid="publish-btn"]');

    await expect(page.locator('[data-testid="success-toast"]'))
      .toContainText('Published successfully');
    await expect(page.locator('[data-testid="listing-status"]'))
      .toHaveText('Active');
  });

  test('shows validation error for missing required fields', async ({ page }) => {
    const dashboard = new AdDashboard(page);
    await dashboard.goto();
    await dashboard.clickCreateAd();
    await page.click('[data-testid="publish-btn"]');

    await expect(page.locator('.field-error')).toHaveCount(2);
  });

});
playwright-test-generator / self-heal.ts (illustrative) 
/**
 * quality-pack: playwright-test-generator
 * Self-healing loop — classifies failures and applies targeted fixes.
 * Runs up to MAX_ITERATIONS before surfacing a human-readable diagnosis.
 */

const MAX_ITERATIONS = 3;

type FailureCategory =
  | 'locator_not_found'
  | 'assertion_mismatch'
  | 'navigation_timeout'
  | 'element_not_interactable'
  | 'test_timeout'
  | 'unrecoverable';

async function classifyFailure(output: string): Promise<FailureCategory> {
  if (output.includes('locator.click') || output.includes('No element found'))
    return 'locator_not_found';
  if (output.includes('expect(received).toBe'))
    return 'assertion_mismatch';
  if (output.includes('Navigation timeout'))
    return 'navigation_timeout';
  return 'unrecoverable';
}

async function healLocator(specPath: string, failOutput: string) {
  // Use Playwright MCP to inspect the live DOM and find the correct selector
  const snapshot = await browserSnapshot();
  const suggestion = await askLLM(`
    The following locator failed: ${failOutput}
    Here is the current DOM snapshot: ${snapshot}
    Suggest the correct Playwright locator (prefer getByRole/getByLabel).
  `);
  await replaceLocatorInFile(specPath, suggestion);
}

async function runWithHealing(specPath: string): Promise<void> {
  for (let i = 1; i <= MAX_ITERATIONS; i++) {
    const { passed, output } = await runPlaywright(specPath);
    if (passed) { console.log(`All tests passed on iteration ${i}`); return; }

    const category = await classifyFailure(output);
    console.log(`[Iteration ${i}] Failure: ${category} — applying fix...`);

    if (category === 'locator_not_found')   await healLocator(specPath, output);
    else if (category === 'unrecoverable')  { reportDiagnosis(output); return; }
  }
  reportDiagnosis('Max healing iterations reached');
}
.github/workflows/coverage-report.yml 
name: Unit Test Coverage Report

on:
  push:
    branches: [main, production]
  pull_request:

jobs:
  coverage:
    runs-on: ubuntu-latest

    steps:
      - uses: actions/checkout@v4

      - name: Run unit tests with coverage
        run: vendor/bin/phpunit --coverage-clover coverage.xml

      - name: Extract coverage percentage
        id: cov
        run: |
          COVERAGE=$(python3 -c "
          import xml.etree.ElementTree as ET
          tree = ET.parse('coverage.xml')
          metrics = tree.find('.//project/metrics')
          covered = int(metrics.get('coveredstatements', 0))
          total   = int(metrics.get('statements', 1))
          print(f'{covered / total * 100:.1f}')
          ")
          echo "pct=$COVERAGE" >> $GITHUB_OUTPUT

      - name: Post coverage comment on PR
        if: github.event_name == 'pull_request'
        uses: actions/github-script@v7
        with:
          script: |
            const pct = '${{ steps.cov.outputs.pct }}';
            const icon = parseFloat(pct) >= 10 ? '✅' : '⚠️';
            await github.rest.issues.createComment({
              issue_number: context.issue.number,
              owner: context.repo.owner,
              repo:  context.repo.repo,
              body:  `${icon} **Unit Test Coverage:** ${pct}%`,
            });

      - name: Upload coverage badge to Pages
        if: github.ref == 'refs/heads/production'
        run: echo "${{ steps.cov.outputs.pct }}" > coverage-latest.txt
.github/workflows/ai-review-gate.yml 
name: AI Review Gate

on:
  pull_request:
    types: [opened, synchronize, reopened]

jobs:
  check-ai-comments:
    name: Block merge if AI reviewer comments unresolved
    runs-on: ubuntu-latest

    steps:
      - name: Check for unresolved AI review comments
        uses: actions/github-script@v7
        with:
          script: |
            const { data: comments } = await github.rest.pulls
              .listReviewComments({
                owner:       context.repo.owner,
                repo:        context.repo.repo,
                pull_number: context.issue.number,
              });

            // Identify comments from AI / bot reviewers
            const aiComments = comments.filter(c =>
              c.user.type === 'Bot'
            );

            // Unresolved = no reply thread started
            const unresolved = aiComments.filter(c =>
              !c.in_reply_to_id
            );

            if (unresolved.length > 0) {
              core.setFailed(
                `${unresolved.length} unresolved AI comment(s). ` +
                `Please address all before merging.`
              );
            } else {
              console.log('All AI comments resolved. Ready to merge.');
            }
