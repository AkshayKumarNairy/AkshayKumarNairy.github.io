// how I think about quality

QA Philosophy

These four principles guide every test strategy, automation decision, and team conversation I have. They're what separates a senior QA mindset from just writing tests.

01 // SHIFT LEFT

Catch it before it's built wrong

I review requirements and designs for testability before a single line of code is written. A bug found in a Jira ticket costs minutes. Found in production, it costs thousands of listings going offline.

02 // AUTOMATE WHAT'S REPETITIVE

Free your brain for the hard stuff

Regression suites, nightly runs, coverage reports, PR gates — all automated. That frees up test capacity for exploratory testing, edge cases, and the scenarios no script would think to check.

03 // QUALITY IS A TEAM SPORT

QA doesn't end at the QA team

I run bi-weekly exploratory sessions with devs, write testability guidelines, and build tools the whole team uses. Quality is a shared responsibility — my job is to make it easy for everyone to participate.

04 // METRICS OVER OPINIONS

If you can't measure it, you can't improve it

Pass rate trends, coverage deltas, bug escape rates, test execution time — I track them all. Decisions about what to automate, what to drop, and where to focus are driven by data, not gut feel.

// ── Page Object ──────────────────────────────────────────────
import { Page, Locator } from '@playwright/test';
import { BasePage } from './BasePage';

export class AdDashboardPage extends BasePage {
  readonly createAdBtn:    Locator;
  readonly titleInput:     Locator;
  readonly rateInput:      Locator;
  readonly categorySelect: Locator;
  readonly publishBtn:     Locator;
  readonly successToast:   Locator;
  readonly listingStatus:  Locator;

  constructor(page: Page) {
    super(page);
    this.createAdBtn    = page.getByRole('button', { name: 'Create listing' });
    this.titleInput     = page.getByLabel('Listing title');
    this.rateInput      = page.getByLabel('Monthly rate');
    this.categorySelect = page.getByLabel('Category');
    this.publishBtn     = page.getByRole('button', { name: 'Publish' });
    this.successToast   = page.getByRole('status');
    this.listingStatus  = page.getByTestId('listing-status');
  }

  async createListing(title: string, rate: string, category: string) {
    await this.createAdBtn.click();
    await this.titleInput.fill(title);
    await this.rateInput.fill(rate);
    await this.categorySelect.selectOption(category);
    await this.publishBtn.click();
  }
}

playwright-test-generator / self-heal.ts (illustrative) 
/**
 * quality-pack: playwright-test-generator
 * Self-healing loop — classifies failures and applies targeted fixes.
 * Runs up to MAX_ITERATIONS before surfacing a human-readable diagnosis.
 */

const MAX_ITERATIONS = 3;

type FailureCategory =
  | 'locator_not_found'
  | 'assertion_mismatch'
  | 'navigation_timeout'
  | 'element_not_interactable'
  | 'test_timeout'
  | 'unrecoverable';

async function classifyFailure(output: string): Promise<FailureCategory> {
  if (output.includes('locator.click') || output.includes('No element found'))
    return 'locator_not_found';
  if (output.includes('expect(received).toBe'))
    return 'assertion_mismatch';
  if (output.includes('Navigation timeout'))
    return 'navigation_timeout';
  return 'unrecoverable';
}

async function healLocator(specPath: string, failOutput: string) {
  // Use Playwright MCP to inspect the live DOM and find the correct selector
  const snapshot = await browserSnapshot();
  const suggestion = await askLLM(`
    The following locator failed: ${failOutput}
    Here is the current DOM snapshot: ${snapshot}
    Suggest the correct Playwright locator (prefer getByRole/getByLabel).
  `);
  await replaceLocatorInFile(specPath, suggestion);
}

async function runWithHealing(specPath: string): Promise<void> {
  for (let i = 1; i <= MAX_ITERATIONS; i++) {
    const { passed, output } = await runPlaywright(specPath);
    if (passed) { console.log(`All tests passed on iteration ${i}`); return; }

    const category = await classifyFailure(output);
    console.log(`[Iteration ${i}] Failure: ${category} — applying fix...`);

    if (category === 'locator_not_found')   await healLocator(specPath, output);
    else if (category === 'unrecoverable')  { reportDiagnosis(output); return; }
  }
  reportDiagnosis('Max healing iterations reached');
}
.github/workflows/coverage-report.yml 
name: Unit Test Coverage Report

on:
  push:
    branches: [main, production]
  pull_request:

jobs:
  coverage:
    runs-on: as24-base-arm

    steps:
      - uses: actions/checkout@v4

      - name: Run unit tests with coverage
        run: vendor/bin/phpunit --coverage-clover coverage.xml

      - name: Extract coverage percentage
        id: cov
        run: |
          COVERAGE=$(python3 -c "
          import xml.etree.ElementTree as ET
          tree = ET.parse('coverage.xml')
          metrics = tree.find('.//project/metrics')
          covered = int(metrics.get('coveredstatements', 0))
          total   = int(metrics.get('statements', 1))
          print(f'{covered / total * 100:.1f}')
          ")
          echo "pct=$COVERAGE" >> $GITHUB_OUTPUT

      - name: Post coverage comment on PR
        if: github.event_name == 'pull_request'
        uses: actions/github-script@v7
        with:
          script: |
            const pct = '${{ steps.cov.outputs.pct }}';
            const icon = parseFloat(pct) >= 10 ? '✅' : '⚠️';
            await github.rest.issues.createComment({
              issue_number: context.issue.number,
              owner: context.repo.owner,
              repo:  context.repo.repo,
              body:  `${icon} **Unit Test Coverage:** ${pct}%`,
            });

      - name: Upload coverage badge to Pages
        if: github.ref == 'refs/heads/production'
        run: echo "${{ steps.cov.outputs.pct }}" > coverage-latest.txt
.github/workflows/ai-review-gate.yml 
name: AI Review Gate

on:
  pull_request:
    types: [opened, synchronize, reopened]

jobs:
  check-ai-comments:
    name: Block merge if AI reviewer comments unresolved
    runs-on: as24-base-arm

    steps:
      - name: Check for unresolved AI review comments
        uses: actions/github-script@v7
        with:
          script: |
            const { data: comments } = await github.rest.pulls
              .listReviewComments({
                owner:       context.repo.owner,
                repo:        context.repo.repo,
                pull_number: context.issue.number,
              });

            // Identify comments from AI / bot reviewers
            const aiComments = comments.filter(c =>
              c.user.type === 'Bot'
            );

            // Unresolved = no reply thread started
            const unresolved = aiComments.filter(c =>
              !c.in_reply_to_id
            );

            if (unresolved.length > 0) {
              core.setFailed(
                `${unresolved.length} unresolved AI comment(s). ` +
                `Please address all before merging.`
              );
            } else {
              console.log('All AI comments resolved. Ready to merge.');
            }
